Вот SQL для двух новых RPC-функций. Пожалуйста, примени их в своей базе данных Supabase.

**SQL для `get_available_materials`:**

```sql
CREATE OR REPLACE FUNCTION public.get_available_materials(p_user_id UUID)
RETURNS SETOF materials
LANGUAGE plpgsql
AS $$
BEGIN
    -- Возвращаем материалы, для которых у активного тарифа пользователя есть доступ,
    -- или материалы, для которых вообще не заданы ограничения по тарифам (доступны всем).
    RETURN QUERY
    SELECT m.*
    FROM public.materials m
    WHERE
        -- Либо для материала нет вообще никаких ограничений по тарифам
        NOT EXISTS (SELECT 1 FROM public.tariff_material_access tma WHERE tma.material_id = m.id)
        -- Либо у пользователя есть активный тариф, которому разрешен доступ к этому материалу
        OR EXISTS (
            SELECT 1
            FROM public.user_tariffs ut
            JOIN public.tariff_material_access tma ON ut.tariff_id = tma.tariff_id
            WHERE ut.user_id = p_user_id
              AND ut.is_active = true
              AND tma.material_id = m.id
        )
    ORDER BY m.order_num;
END;
$$;
```

**SQL для `get_available_chats`:**

```sql
CREATE OR REPLACE FUNCTION public.get_available_chats(p_user_id UUID)
RETURNS SETOF chats
LANGUAGE plpgsql
AS $$
BEGIN
    -- Логика аналогична материалам.
    RETURN QUERY
    SELECT c.*
    FROM public.chats c
    WHERE
        -- Либо для чата нет вообще никаких ограничений по тарифам
        NOT EXISTS (SELECT 1 FROM public.tariff_chat_access tca WHERE tca.chat_id = c.id)
        -- Либо у пользователя есть активный тариф, которому разрешен доступ к этому чату
        OR EXISTS (
            SELECT 1
            FROM public.user_tariffs ut
            JOIN public.tariff_chat_access tca ON ut.tariff_id = tca.tariff_id
            WHERE ut.user_id = p_user_id
              AND ut.is_active = true
              AND tca.chat_id = c.id
        )
    ORDER BY c.order_num;
END;
$$;
```
